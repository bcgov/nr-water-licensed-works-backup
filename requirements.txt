# Keep this list short. Anything added needs a reason.
#
# Versions are pinned. These jobs run unattended on a schedule, and an
# unpinned dependency means a run that worked on Friday can fail on Monday
# for reasons unrelated to this code - which surfaces as SYSTEM_FAIL and
# wastes an investigation. Bump deliberately, and re-run the tests.
#
# Only the direct dependencies are pinned, not the ~70 packages arcgis
# pulls in transitively. Pinning those too would need a lock file, which is
# more machinery than this project warrants.
#
# jenkins/notify.py deliberately uses only boto3 and the standard library,
# because the GIS server it runs on has limited Python environments. It
# installs from jenkins/requirements.txt, not this file, so that arcgis is
# never installed on that server.

arcgis==2.4.3
boto3==1.43.71
PyYAML==6.0.3

# Reads the feature count out of the exported File Geodatabase, which is how
# backup.py proves the artifact is a faithful snapshot rather than merely an
# intact zip (DESIGN.md 6.5 step 2). arcpy is not available on a GitHub
# runner and nothing else in the dependency set can open an FGDB. pyogrio
# ships GDAL wheels, so this needs no system GDAL install; only backup.py
# imports it, and checks.py must never do so.
pyogrio==0.13.0

# Testing only
pytest==9.1.1
